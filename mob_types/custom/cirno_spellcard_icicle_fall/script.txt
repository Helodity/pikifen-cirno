first_state = idling

init {
    set_var spawn_rate 0.55
    set_var turn_amt 90
}

script {
    idling {
        on_enter {
            set_timer 1
            set_shadow_visibility false
        }
        on_timer {
            spawn projectile_l1
            spawn projectile_r1

            spawn projectile_l2
            spawn projectile_r2

            spawn projectile_l3
            spawn projectile_r3

            //All projectiles are spawned with "icicle_fall_setup" set to true, find them, and give them their speed modifier.
            get_mob_ids_with_var mobs_to_setup icicle_fall_setup true
            for_each it mob_id $mobs_to_setup
                focus_on_id $mob_id

                get_focus_var p_invert icicle_fall_invert_turn

                set_var tmp_turn_amt $turn_amt
                if $p_invert = true
                    calculate tmp_turn_amt $tmp_turn_amt * -1
                end_if

                set_focus_var turn_amount $tmp_turn_amt

                get_focus_var p_speed icicle_fall_speed
                if $p_speed = 1
                    run_next_action_as focus
                    receive_status cirno_projectile_speed_1
                end_if
                if $p_speed = 2
                    run_next_action_as focus
                    receive_status cirno_projectile_speed_2
                end_if
                if $p_speed = 3
                    run_next_action_as focus
                    receive_status cirno_projectile_speed_3
                end_if

                set_focus_var icicle_fall_setup false

            end_for_each
            calculate turn_amt $turn_amt + 5

            if $turn_amt > 145
                calculate turn_amt $turn_amt - 55
            end_if

            set_timer $spawn_rate
        }

        on_tick {
            get_mob_ids_with_var mobs_to_setup icicle_fall_setup false

            //Check for projectiles that have started turning, and set their projectile speed to be slower
            for_each it mob_id $mobs_to_setup
                focus_on_id $mob_id
                get_focus_var p_speed icicle_fall_speed
                get_focus_var p_turn_delay turn_delay

                if $p_turn_delay <= 0
                    if $p_speed = 1
                        run_next_action_as focus
                        remove_status cirno_projectile_speed_1
                    end_if
                    if $p_speed = 2
                        run_next_action_as focus
                        remove_status cirno_projectile_speed_2
                    end_if
                    if $p_speed = 3
                        run_next_action_as focus
                        remove_status cirno_projectile_speed_3
                    end_if
                    receive_status cirno_projectile_speed_1
                    set_focus_var icicle_fall_setup done
                end_if

            end_for_each
        }
    }
}
